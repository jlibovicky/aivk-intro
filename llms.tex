\begin{frame}{Generativní jazykové modely}

    \begin{itemize}[<+->]

        \item GPT = Generative Pre-trained Transformer

        \item Model, který předpovídá, jaké slovo v textu může následovat \\

        \item GPT-3 je natrénovaný na 45TB textu (37 milionů Zločinů a trestů),
            má 175 miliard parametrů (1600$\times$ více než BERT)

            \begin{itemize}[<+->]

                \item 37M Zločinů a trestů by na ploše fotbalového hřiště
                    dosáhlo výšky 4.5m

                \item BERT by mohl běžet na GPU z~PlayStation 5

                \item GPT-3 by potřebovalo přes 100 PlayStationů

            \end{itemize}

\begin{frame}{AI Asistent}

    \begin{itemize}[<+->]

        \item Vnější algoritmus volá LLM: prosí ho o další token, dokud nedokončí odpověď

        \item Může interpretovat některé výstupy LLM jako příkazy (typicky
            non-AI programy), které vykoná a pak pošle výsledek zpět LLM (agenti)

        \item RAG -- Retrieval Augmented Generation: LLM může vyhledávat
            informace v~externí databázi a pak je použít pro generování odpovědi

    \end{itemize}

\end{frame}

\begin{frame}{Limity LLM}

    \begin{itemize}[<+->]

        \item \textbf{Halucinace}: model sebejistě vymýšlí fakta, citace,
            zdroje -- trénování odměňuje odpověď víc než přiznání nejistoty
            {\footnotesize\citep{kalai2025hallucinate}}

        \item \textbf{Podlézavost} (sycophancy): model souhlasí s~uživatelem
            a lichotí mu {\footnotesize\citep{sharma2024sycophancy}}

        \item \textbf{Datum znalostí}: model ví jen to, co bylo v~trénovacích
            datech, novější informace jen přes vyhledávání

        \item \textbf{Zdroje}: model sám neví, odkud informaci má; citace
            fungují jen s~vyhledáváním (a i tak ne vždy správně)

        \item Výkon je \textbf{nerovnoměrný}: vyřeší olympiádní úlohu, ale
            splete se v~jednoduché věci

    \end{itemize}

\end{frame}

    \end{itemize}

    \begin{itemize}[<+->]

            srpna 2025, další postupně)

    \end{itemize}
